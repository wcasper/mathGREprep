\section{Calculus Problems}

\subsection{Standard problems}


\begin{frame}{Problem 1}
Suppose that $f(x)$ is twice differentiable with $f(2) = 10$ and $f'(x) = x^2f(x)$.
Find $f''(2)$.
\end{frame}

\begin{frame}{Problem 2}
Find
$$\lim_{x\rightarrow 0}\frac{\sin(3x)\sin(2x)}{x\tan(6x)}.$$
\end{frame}

\begin{frame}{Problem 3}
How many tangent lines to the curve $y=x^2$ pass through the point $(3,-7)$ and what are their slopes?
\end{frame}

\begin{frame}{Problem 4}
The tangent line to the curve $y = f(x)$ at $x=2$ is $y-8 = 2(x-2)$.  Calculate
$$\lim_{x\rightarrow 2}\frac{xf(2)-2f(x)}{x-2}$$
\end{frame}

\begin{frame}{Problem 5}
Suppose that $f(x) = |x-1|-|x+1|$.  Compute $\int_{-3}^3(f(x-1)-f(x+1)+1)dx$
\end{frame}

\begin{frame}{Problem 6}
Find an integral expression which is equal to
{\small
$$\lim_{n\rightarrow\infty}\frac{3}{n^2}\left(\sqrt{4n^2 + 9\times 1^2} + \sqrt{4n^2 + 9\times 2^2} + \dots + \sqrt{4n^2 + 9\times n^2}\right)$$
}
\end{frame}

\begin{frame}{Problem 7}
A region is bounded by two concentric circles, where the smaller circle has radius $r$ and the bigger circle has radius $R$.
If $r$ decreases at $1$ inch per second and $R$ increases at $2$ inches per second, how fast is the area changing at the instant when $r=3$ inches and $R=4$ inches?
\end{frame}

\begin{frame}{Problem 8}
Let $f(x) = x^{2024}$.  Evaluate
$$f(1) + \frac{f'(1)}{1!} + \frac{f''(1)}{2!} + \dots + \frac{f^{(2024)}(1)}{2024!}$$
\end{frame}

\begin{frame}{Problem 9}
Find the smallest positive integer $m$ such that $m^{1/m} \geq n^{1/n}$ for all positive integers $n\geq m$.
\end{frame}

\begin{frame}{Problem 10}
Find all values of $x$ such that
$$\lim_{n\rightarrow\infty} (x + x^3 + x^5 + \dots + x^{2n-1}) = \frac{3}{8}.$$
\end{frame}
